def write_json(payload, exit_code=0):
    data = json.dumps(payload, ensure_ascii=False)
    sys.stdout.write(data)
    sys.stdout.write("\n")
    sys.stdout.flush()
    sys.exit(exit_code)


def extract_text_layer_first(pdf_path: Path):
    doc = fitz.open(str(pdf_path))
    pages = []

    for page_index in range(len(doc)):
        page = doc.load_page(page_index)
        text = (page.get_text("text") or "").strip()

        if text:
            pages.append({
                "page": page_index + 1,
                "text": text,
            })

    doc.close()

    final_text = "\n\n".join(
        f"===== PAGE {item['page']} =====\n{item['text']}"
        for item in pages
    ).strip()

    return pages, final_text


def render_pdf_pages(pdf_path: Path, output_dir: Path, dpi: int, max_pages: int = 0):
    doc = fitz.open(str(pdf_path))
    total_pages = len(doc)

    if max_pages and max_pages > 0:
        page_limit = min(total_pages, max_pages)
    else:
        page_limit = total_pages

    image_paths = []

    for page_index in range(page_limit):
        page = doc.load_page(page_index)
        pix = page.get_pixmap(dpi=dpi, alpha=False)

        image_path = output_dir / f"page_{page_index + 1}.png"
        pix.save(str(image_path))

        image_paths.append(image_path)

    doc.close()
    return total_pages, image_paths

def extract_paddle_text(result):
    lines = []

    if not isinstance(result, list):
        return ""

    for page_result in result:
        if not isinstance(page_result, list):
            continue

        for item in page_result:
            try:
                text = str(item[1][0]).strip()
                if text:
                    lines.append(text)
            except Exception:
                pass

    return "\n".join(lines).strip()


def run_fast_page(ocr, image_path: Path):
    result = ocr.ocr(str(image_path), cls=False)
    return extract_paddle_text(result)

        image_cv = cv2.imread(str(image_path))

        with contextlib.redirect_stdout(sys.stderr):
            det_output = detector.text_detector(image_cv)

        lines = []

        boxes = []
        if isinstance(det_output, tuple):
            boxes = det_output[0]
        else:
            boxes = det_output

        if boxes is None:
            boxes = []

        boxes = sorted(
            boxes,
            key=lambda box: (
                min(p[1] for p in box),
                min(p[0] for p in box),
            ),
        )

        for box in boxes:
            try:
                xs = [p[0] for p in box]
                ys = [p[1] for p in box]

                left = max(0, int(min(xs)) - 3)
                top = max(0, int(min(ys)) - 3)
                right = min(image.width, int(max(xs)) + 3)
                bottom = min(image.height, int(max(ys)) + 3)

                if right <= left or bottom <= top:
                    continue

                crop = image.crop((left, top, right, bottom))
                text = recognizer.predict(crop).strip()

                if text:
                    lines.append(text)
            except Exception:
                continue

        text = "\n".join(lines).strip()
        seconds = round(time.time() - page_started, 2)

        page_texts.append({
            "page": page_number,
            "text": text,
            "seconds": seconds,
        })

        print(
            f"QUALITY PAGE {page_number}/{len(image_paths)} done in {seconds}s",
            file=sys.stderr,
            flush=True,
        )

    return page_texts

import json

def _find_json_payload(stdout: str) -> dict[str, Any]:
    lines = stdout.splitlines()

    for line in reversed(lines):
        line = line.strip()

        if not line:
            continue

        if line.startswith("{") and line.endswith("}"):
            return json.loads(line)

    raise RuntimeError(
        "Worker khĂ´ng tráº£ JSON há»£p lá»‡. STDOUT: "
        + stdout[-2000:]
    )


def run_paddle_ocr_worker(
    pdf_path: str,
    language: str = "vi",
    dpi: int = 300,
    timeout_seconds: int = 1800,
    vietocr_model: str = "vgg_transformer",
) -> dict[str, Any]:
    base_dir = Path(settings.BASE_DIR)

    worker_python = base_dir / ".venv_ocr_paddle" / "Scripts" / "python.exe"
    worker_script = base_dir / "scripts" / "paddle_ocr_worker.py"

    if not worker_python.exists():
        raise RuntimeError(f"KhĂ´ng tĂ¬m tháº¥y Python OCR worker: {worker_python}")

    if not worker_script.exists():
        raise RuntimeError(f"KhĂ´ng tĂ¬m tháº¥y script OCR worker: {worker_script}")

    pdf_file = Path(pdf_path)

    if not pdf_file.exists():
        raise RuntimeError(f"PDF khĂ´ng tá»“n táº¡i: {pdf_file}")

    env = os.environ.copy()
    env["KMP_DUPLICATE_LIB_OK"] = "TRUE"
    env["FLAGS_use_mkldnn"] = "0"
    env["OMP_NUM_THREADS"] = "1"
    env["PYTHONUTF8"] = "1"
    env["PYTHONIOENCODING"] = "utf-8"

    command = [
        str(worker_python),
        str(worker_script),
        "--pdf",
        str(pdf_file),
        "--language",
        language or "vi",
        "--dpi",
        str(dpi or 300),
        "--vietocr-model",
        vietocr_model or "vgg_transformer",
    ]

    completed = subprocess.run(
        command,
        cwd=str(base_dir),
        env=env,
        capture_output=True,
        text=True,
        encoding="utf-8",
        errors="replace",
        timeout=timeout_seconds,
    )

    if completed.returncode != 0:
        raise RuntimeError(
            "Paddle/VietOCR worker cháº¡y lá»—i. "
            f"Return code: {completed.returncode}\n"
            f"STDOUT:\n{completed.stdout[-4000:]}\n"
            f"STDERR:\n{completed.stderr[-4000:]}"
        )

    payload = _find_json_payload(completed.stdout)

    if not payload.get("success"):
        raise RuntimeError(
            payload.get("error")
            or "Worker OCR tráº£ success=false nhÆ°ng khĂ´ng cĂ³ error."
        )

    return payload

def collect_text_from_paddle_result(result: Any) -> list[str]:
    """
    PaddleOCR cĂ³ nhiá»u format result giá»¯a cĂ¡c version.

    HĂ m nĂ y cá»‘ gáº¯ng Ä‘á»c text tá»« cáº£:
    - result kiá»ƒu list cá»• Ä‘iá»ƒn
    - result kiá»ƒu dict má»›i cĂ³ rec_texts
    - object cĂ³ .json hoáº·c .res
    """

    texts: list[str] = []

    def walk(value: Any):
        if value is None:
            return

        if isinstance(value, str):
            return

        if isinstance(value, dict):
            if isinstance(value.get("rec_texts"), list):
                for text in value["rec_texts"]:
                    if isinstance(text, str) and text.strip():
                        texts.append(text.strip())

            if isinstance(value.get("texts"), list):
                for text in value["texts"]:
                    if isinstance(text, str) and text.strip():
                        texts.append(text.strip())

            if isinstance(value.get("text"), str) and value["text"].strip():
                texts.append(value["text"].strip())

            for item in value.values():
                walk(item)

            return

        if isinstance(value, (list, tuple)):
            # Format cá»• Ä‘iá»ƒn:
            # [ box, ("text", score) ]
            if (
                len(value) >= 2
                and isinstance(value[1], (list, tuple))
                and len(value[1]) >= 1
                and isinstance(value[1][0], str)
            ):
                text = value[1][0].strip()

                if text:
                    texts.append(text)

                return

            # Format:
            # ("text", score)
            if (
                len(value) >= 2
                and isinstance(value[0], str)
                and isinstance(value[1], (int, float))
            ):
                text = value[0].strip()

                if text:
                    texts.append(text)

                return

            for item in value:
                walk(item)

            return

        # PaddleOCR 3.x cĂ³ thá»ƒ tráº£ object cĂ³ json/res
        for attr_name in ["json", "res"]:
            if hasattr(value, attr_name):
                try:
                    walk(getattr(value, attr_name))
                    return
                except Exception:
                    pass

        if hasattr(value, "to_dict"):
            try:
                walk(value.to_dict())
                return
            except Exception:
                pass

    walk(result)

    # Loáº¡i trĂ¹ng nhÆ°ng giá»¯ thá»© tá»±
    unique_texts = []
    seen = set()

    for text in texts:
        if text not in seen:
            unique_texts.append(text)
            seen.add(text)

    return unique_texts


def run_paddleocr_on_image(image_path: str, language: str = "vi") -> list[str]:
    """
    Cháº¡y PaddleOCR trĂªn má»™t áº£nh.
    """

    ocr = get_paddle_ocr_engine(language)

    # PaddleOCR báº£n má»›i thÆ°á»ng cĂ³ predict()
    if hasattr(ocr, "predict"):
        try:
            result = ocr.predict(input=image_path)
            texts = collect_text_from_paddle_result(result)

            if texts:
                return texts
        except Exception:
            pass

    # PaddleOCR báº£n cÅ© thÆ°á»ng dĂ¹ng ocr()
    if hasattr(ocr, "ocr"):
        try:
            try:
                result = ocr.ocr(image_path, cls=True)
            except TypeError:
                result = ocr.ocr(image_path)

            texts = collect_text_from_paddle_result(result)

            if texts:
                return texts
        except Exception as exc:
            raise RuntimeError(f"PaddleOCR Ä‘á»c áº£nh lá»—i: {exc}") from exc

    return []


def extract_text_by_paddleocr(
    pdf_path: str,
    language: str = "vi",
    zoom: float = 1.8,
    max_pages: Optional[int] = None,
) -> ExtractResult:
    """
    OCR PDF scan áº£nh báº±ng PaddleOCR v2 cháº¡y á»Ÿ mĂ´i trÆ°á»ng riĂªng.

    Backend chĂ­nh khĂ´ng import PaddleOCR trá»±c tiáº¿p ná»¯a.
    Django sáº½ gá»i:
        .venv_ocr_paddle/Scripts/python.exe scripts/paddle_ocr_worker.py

    CĂ¡ch nĂ y trĂ¡nh lá»—i PaddleOCR 3.x / oneDNN / PIR trĂªn Windows.
    """

    dpi = 180

    try:
        if zoom and zoom > 0:
            dpi = max(150, min(220, int(zoom * 100)))
    except Exception:
        dpi = 180

    worker_result = run_paddle_ocr_worker(
        pdf_path=pdf_path,
        language=language or "vi",
        dpi=dpi,
        timeout_seconds=900,
    )

    extracted_text = (worker_result.get("text") or "").strip()
    page_count = int(worker_result.get("page_count") or 0)
    engine = (
        worker_result.get("engine")
        or f"pymupdf+paddleocr-v2-subprocess:{language}"
    )

    warning = ""

    if not extracted_text:
        warning = (
            "PaddleOCR worker cháº¡y xong nhÆ°ng khĂ´ng nháº­n Ä‘Æ°á»£c text. "
            "CĂ³ thá»ƒ file scan má» hoáº·c cháº¥t lÆ°á»£ng áº£nh quĂ¡ tháº¥p."
        )

    return ExtractResult(
        text=extracted_text,
        page_count=page_count,
        engine=engine,
        warning=warning,
    )

def extract_text_from_pdf(
    pdf_path: str,
    language: str = "vi",
    force_paddleocr: bool = False,
) -> ExtractResult:
    """
    OCR thĂ´ng minh:

    force_paddleocr=False:
        1. Thá»­ PyMuPDF trÆ°á»›c.
        2. Náº¿u PDF khĂ´ng cĂ³ text thĂ¬ dĂ¹ng PaddleOCR v2 worker.

    force_paddleocr=True:
        Bá» qua text sáºµn, luĂ´n render trang vĂ  cháº¡y PaddleOCR v2 worker.
    """

    if not force_paddleocr:
        direct_result = extract_text_by_pymupdf(pdf_path)

        if direct_result.text.strip():
            return direct_result

    try:
        paddle_result = extract_text_by_paddleocr(
            pdf_path=pdf_path,
            language=language,
        )

        if paddle_result.text.strip():
            return paddle_result

        if not paddle_result.warning:
            paddle_result.warning = (
                "PaddleOCR worker cháº¡y xong nhÆ°ng khĂ´ng nháº­n Ä‘Æ°á»£c text. "
                "CĂ³ thá»ƒ file scan má» hoáº·c model chÆ°a phĂ¹ há»£p."
            )

        return paddle_result

    except Exception as exc:
        if force_paddleocr:
            return ExtractResult(
                text="",
                page_count=0,
                engine=f"pymupdf+paddleocr-v2-subprocess:{language}",
                warning=str(exc),
            )

        direct_result = extract_text_by_pymupdf(pdf_path)

        direct_result.warning = (
            "PDF khĂ´ng cĂ³ text sáºµn hoáº·c OCR áº£nh tháº¥t báº¡i báº±ng PaddleOCR v2 worker. "
            f"Chi tiáº¿t: {exc}"
        )

        return direct_result

def run_ocr_for_digital_file(
    digital_file: DigitalFile,
    user=None,
    language: str = "vi",
    force: bool = False,
    force_paddleocr: bool = False,
) -> OcrJob:
    """
    Cháº¡y OCR cho DigitalFile.

    force=False:
        Náº¿u Ä‘Ă£ cĂ³ job COMPLETED cĂ³ text thĂ¬ tráº£ láº¡i job má»›i nháº¥t.

    force=True:
        Cháº¡y láº¡i OCR vĂ  táº¡o job má»›i.
    """

    if not force:
        existing_job = (
            OcrJob.objects
            .filter(
                digital_file=digital_file,
                status=OcrJob.Status.COMPLETED,
            )
            .exclude(extracted_text="")
            .order_by("-created_at", "-id")
            .first()
        )

        if existing_job:
            return existing_job

    job = OcrJob.objects.create(
        digital_file=digital_file,
        profile=digital_file.profile,
        document=digital_file.document,
        created_by=user if user and user.is_authenticated else None,
        status=OcrJob.Status.PENDING,
        language=language,
    )

    job.status = OcrJob.Status.RUNNING
    job.started_at = timezone.now()
    job.save(update_fields=["status", "started_at", "updated_at"])

    try:
        if getattr(digital_file, "is_deleted", False):
            raise ValueError("File sá»‘ hĂ³a Ä‘Ă£ bá»‹ xĂ³a má»m, khĂ´ng thá»ƒ OCR.")

        if not is_pdf_file(digital_file):
            raise ValueError("OCR hiá»‡n chá»‰ há»— trá»£ file PDF.")

        file_path = get_file_path(digital_file)

        if not file_path:
            raise FileNotFoundError("KhĂ´ng láº¥y Ä‘Æ°á»£c Ä‘Æ°á»ng dáº«n file váº­t lĂ½.")

        if not os.path.exists(file_path):
            raise FileNotFoundError(f"File váº­t lĂ½ khĂ´ng tá»“n táº¡i: {file_path}")

        result = extract_text_from_pdf(
            pdf_path=file_path,
            language=language,
            force_paddleocr=force_paddleocr,
        )

        extracted_text = result.text.strip()
        normalized_text = normalize_search_text(extracted_text)

        if not extracted_text and result.warning:
            job.status = OcrJob.Status.FAILED
            job.engine = result.engine
            job.page_count = result.page_count
            job.character_count = 0
            job.extracted_text = ""
            job.normalized_text = ""
            job.error_message = result.warning
            job.finished_at = timezone.now()
            job.save(
                update_fields=[
                    "status",
                    "engine",
                    "page_count",
                    "character_count",
                    "extracted_text",
                    "normalized_text",
                    "error_message",
                    "finished_at",
                    "updated_at",
                ]
            )

            return job

        job.status = OcrJob.Status.COMPLETED
        job.engine = result.engine
        job.page_count = result.page_count
        job.character_count = len(extracted_text)
        job.extracted_text = extracted_text
        job.normalized_text = normalized_text
        job.error_message = result.warning or ""
        job.finished_at = timezone.now()
        job.save(
            update_fields=[
                "status",
                "engine",
                "page_count",
                "character_count",
                "extracted_text",
                "normalized_text",
                "error_message",
                "finished_at",
                "updated_at",
            ]
        )

        return job

    except Exception as exc:
        job.status = OcrJob.Status.FAILED
        job.error_message = str(exc)
        job.finished_at = timezone.now()
        job.save(
            update_fields=[
                "status",
                "error_message",
                "finished_at",
                "updated_at",
            ]
        )

        return job

def search_ocr_text(query: str, limit: int = 50):
    normalized_query = normalize_search_text(query or "")

    queryset = (
        OcrJob.objects
        .select_related("digital_file", "profile", "document", "created_by")
        .filter(status=OcrJob.Status.COMPLETED)
        .exclude(normalized_text="")
    )

    tokens = [token for token in normalized_query.split() if token]

    for token in tokens:
        queryset = queryset.filter(normalized_text__icontains=token)

    return queryset.order_by("-created_at", "-id")[:limit]

    permission_classes = [IsAuthenticated, CanViewArchive]

    def get(self, request):
        queryset = (
            OcrJob.objects
            .select_related("digital_file", "profile", "document", "created_by")
            .all()
        )

        q = request.query_params.get("q")
        status_value = request.query_params.get("status")
        mode_value = request.query_params.get("ocr_mode")
        digital_file_id = request.query_params.get("digital_file")

        document_id = request.query_params.get("document")

        if q:
            queryset = queryset.filter(
                Q(extracted_text__icontains=q)
                | Q(normalized_text__icontains=normalize_for_search(q))
                | Q(digital_file__original_name__icontains=q)
                | Q(profile__profile_code__icontains=q)
                | Q(profile__title__icontains=q)
                | Q(document__document_code__icontains=q)
                | Q(document__title__icontains=q)
            )

        if status_value:
            queryset = queryset.filter(status=status_value)

        if mode_value:
            queryset = queryset.filter(ocr_mode=mode_value)

        if digital_file_id:
            queryset = queryset.filter(digital_file_id=digital_file_id)

        if profile_id:
            queryset = queryset.filter(profile_id=profile_id)

        if document_id:
            queryset = queryset.filter(document_id=document_id)

        page = max(int(request.query_params.get("page", 1)), 1)
        page_size = min(max(int(request.query_params.get("page_size", 20)), 1), 100)

        total = queryset.count()
        start = (page - 1) * page_size
        end = start + page_size

        serializer = OcrJobSerializer(queryset[start:end], many=True)

        return success_response(
            message="Lấy danh sách OCR job thành công.",
            data={
                "results": serializer.data,
                "pagination": {
                    "page": page,
                    "page_size": page_size,
                    "total": total,
                    "total_pages": (total + page_size - 1) // page_size,
                    "has_next": end < total,
                    "has_previous": page > 1,
                },
            },
        )

import json
import os
import re
import subprocess
import threading
import unicodedata
from pathlib import Path

from django.conf import settings

from apps.ocr.models import OcrJob


BASE_DIR = Path(settings.BASE_DIR)
OCR_PYTHON = BASE_DIR / ".venv_ocr_paddle" / "Scripts" / "python.exe"
OCR_WORKER = BASE_DIR / "scripts" / "paddle_ocr_worker.py"


def normalize_for_search(text: str) -> str:
    if not text:
        return ""
    text = unicodedata.normalize("NFD", text)
    text = "".join(ch for ch in text if unicodedata.category(ch) != "Mn")
    text = text.replace("Ä‘", "d").replace("Ä", "D")
    text = text.lower()
    text = re.sub(r"[^a-z0-9]+", " ", text)
    return re.sub(r"\s+", " ", text).strip()


def get_pdf_page_count(pdf_path):
    try:
        import fitz
        doc = fitz.open(str(pdf_path))
        total = len(doc)
        doc.close()
        return total
    except Exception:
        return 0


def run_worker_and_update_job(job_id, pdf_path, mode, dpi=120):
    total_pages = get_pdf_page_count(pdf_path)

    OcrJob.objects.filter(id=job_id).update(
        status=OcrJob.Status.RUNNING,
        started_at=timezone.now(),
        error_message="",
        page_count=total_pages,
        current_page=1 if total_pages else 0,
        progress_percent=1,
        updated_at=timezone.now(),
    )

    env = os.environ.copy()
    env["PYTHONIOENCODING"] = "utf-8"
    env["PYTHONUTF8"] = "1"

    command = [
        str(OCR_PYTHON),
        str(OCR_WORKER),
        "--pdf",
        str(pdf_path),
        "--language",
        "vi",
        "--dpi",
        str(dpi),
        "--mode",
        mode,
        "--vietocr-model",
        ("vgg_seq2seq" if mode == "quality" else "vgg_transformer"),
    ]

    stdout_lines = []
    stderr_lines = []

    try:
        process = subprocess.Popen(
            command,
            cwd=str(BASE_DIR),
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            encoding="utf-8",
            errors="replace",
            env=env,
            bufsize=1,
        )

        def read_stderr():
            for line in process.stderr:
                line = line.strip()
                if line:
                    stderr_lines.append(line)

                match = re.search(r"(FAST|QUALITY) PAGE\s+(\d+)/(\d+)\s+done", line)
                if match:
                    done_page = int(match.group(2))
                    total = int(match.group(3))
                    next_page = min(done_page + 1, total)
                    percent = int(done_page * 100 / max(total, 1))

                    OcrJob.objects.filter(id=job_id).update(
                        current_page=next_page,
                        page_count=total,
                        progress_percent=percent,
                        updated_at=timezone.now(),
                    )

        stderr_thread = threading.Thread(target=read_stderr, daemon=True)
        stderr_thread.start()

        for line in process.stdout:
            if line:
                stdout_lines.append(line)

        return_code = process.wait()
        stderr_thread.join(timeout=2)

        if return_code != 0:
            raise RuntimeError(
                "OCR worker failed with code "
                + str(return_code)
                + "\nCOMMAND: "
                + " ".join(command)
                + "\nSTDERR:\n"
                + "\n".join(stderr_lines[-100:])
                + "\nSTDOUT:\n"
                + "\n".join(stdout_lines[-30:])
            )

        json_line = None
        for line in reversed(stdout_lines):
            line = line.strip()
            if line.startswith("{") and line.endswith("}"):
                json_line = line
                break

        if not json_line:
            raise RuntimeError(
                "KhĂ´ng tĂ¬m tháº¥y JSON output tá»« OCR worker.\nSTDERR:\n"
                + "\n".join(stderr_lines[-100:])
            )

        data = json.loads(json_line)

        if not data.get("success"):
            raise RuntimeError(data.get("error") or "OCR worker tráº£ success=false.")

        text = data.get("text") or ""
        page_count = data.get("page_count") or total_pages

        OcrJob.objects.filter(id=job_id).update(
            status=OcrJob.Status.COMPLETED,
            engine=data.get("engine") or f"paddle-{mode}",
            language=data.get("language") or "vi",
            page_count=page_count,
            character_count=len(text),
            extracted_text=text,
            normalized_text=normalize_for_search(text),
            current_page=page_count,
            progress_percent=100,
            error_message="",
            finished_at=timezone.now(),

from django.core.management.base import BaseCommand
from apps.ocr.models import OcrJob
from apps.ocr.runner import run_worker_and_update_job


class Command(BaseCommand):
    help = "Run OCR fast then quality for one digital file."

    def add_arguments(self, parser):
        parser.add_argument("--fast-job-id", type=int, required=True)
        parser.add_argument("--quality-job-id", type=int, required=True)
        parser.add_argument("--pdf", required=True)

    def handle(self, *args, **options):
        fast_job_id = options["fast_job_id"]
        quality_job_id = options["quality_job_id"]
        pdf_path = options["pdf"]

        run_worker_and_update_job(
            job_id=fast_job_id,
            pdf_path=pdf_path,
            mode="fast",
            dpi=120,
        )

        fast_job = OcrJob.objects.get(id=fast_job_id)

        if fast_job.status == OcrJob.Status.COMPLETED:
            run_worker_and_update_job(
                job_id=quality_job_id,
                pdf_path=pdf_path,
                mode="quality",
                dpi=120,
            )
        else:
            OcrJob.objects.filter(id=quality_job_id).update(
                status=OcrJob.Status.FAILED,
                error_message="Quality OCR không chạy vì fast OCR thất bại.",
            )
